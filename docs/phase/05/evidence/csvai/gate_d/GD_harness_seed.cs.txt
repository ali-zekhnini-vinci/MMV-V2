var services = new ServiceCollection()

    .BuildServiceProvider();
var password = File.Exists(credsFile) ? JsonDocument.Parse(File.ReadAllText(credsFile)).RootElement[0].GetProperty("password").GetString()! : "Banc-" + Convert.ToHexString(System.Security.Cryptography.RandomNumberGenerator.GetBytes(12));
var accounts = new List<object>();
using (var scope = services.CreateScope())
{
    var context = scope.ServiceProvider.GetRequiredService<MmvDbContext>();
    var hashing = scope.ServiceProvider.GetRequiredService<IPasswordHashing>();
    var now = DateTimeOffset.UtcNow;
    if (!context.Stores.Any())
    {
        var shop = Store.Create(Guid.CreateVersion7(), "NATIF-GD", "Optique du Banc GATE D (fictif)", "BE", Currencies.Euro, now);
        context.Stores.Add(shop);
        context.Locations.Add(Location.Create(Guid.CreateVersion7(), shop.Id, "MAGASIN", "Magasin", true, now));
    }

    foreach (var (name, display, perms) in new[]
    {
        ("gd-admin", "Admin GATE D", Permissions.Phase05.Append(Permissions.AuditRead).ToArray()),
        ("gd-vendeur", "Vendeur GATE D", new[] { Permissions.CatalogRead, Permissions.StockRead }),
    })
    {
        if (!context.Users.Any(u => u.UserName == name))
        {
            var user = new UserAccount(Guid.CreateVersion7(), name, display, now);
            user.SetPasswordHash(hashing.Hash(user, password));
            foreach (var p in perms)
            {
                user.Grant(p);
            }

            context.Users.Add(user);
        }

        accounts.Add(new { key = name, userName = name, password, displayName = display });
    }

    await context.SaveChangesAsync();
}

async Task<JsonElement> Send(HttpMethod method, string path, object body)
{
    using var request = new HttpRequestMessage(method, path) { Content = JsonContent.Create(body) };
    using var response = await http.SendAsync(request);
    var text = await response.Content.ReadAsStringAsync();
    if (!response.IsSuccessStatusCode)
    {
        throw new InvalidOperationException($"{method} {path} → {(int)response.StatusCode} {text}");
    }

    return text.Length == 0 ? default : JsonDocument.Parse(text).RootElement.Clone();
}
