const string Connection = "Host=127.0.0.1;Port=55433;Database=mmv_v2_scratch;Username=mmv_v2_scratch;Password=scratch";
var api = new Uri(args[0]);
var credsFile = args[1];

    throw new InvalidOperationException("Cible refusée.");

var services = new ServiceCollection()
    .AddInfrastructure(new ConfigurationBuilder().AddInMemoryCollection(new Dictionary<string, string?> { ["ConnectionStrings:Mmv"] = Connection }).Build())
    .BuildServiceProvider();
var Password = "Banc-" + Convert.ToHexString(System.Security.Cryptography.RandomNumberGenerator.GetBytes(12));
if (File.Exists(credsFile))
{
    Password = JsonDocument.Parse(File.ReadAllText(credsFile)).RootElement[0].GetProperty("password").GetString()!;
}

var accounts = new List<object>();
using (var scope = services.CreateScope())
{
    var context = scope.ServiceProvider.GetRequiredService<MmvDbContext>();
    var hashing = scope.ServiceProvider.GetRequiredService<IPasswordHashing>();
    var now = DateTimeOffset.UtcNow;
    if (!context.Stores.Any(s => s.Code == "NATIF-UX"))
    {
        var shop = Store.Create(Guid.CreateVersion7(), "NATIF-UX", "Optique du Banc UX (fictif)", "BE", Currencies.Euro, now);
        context.Stores.Add(shop);
        context.Locations.Add(Location.Create(Guid.CreateVersion7(), shop.Id, "MAGASIN", "Magasin", true, now));
        context.Locations.Add(Location.Create(Guid.CreateVersion7(), shop.Id, "RESERVE", "Réserve", false, now));
    }

    foreach (var (name, display, perms) in new[]
    {
        ("uxai-admin", "Admin Banc UX", Permissions.Phase05.Append(Permissions.AuditRead).ToArray()),
        ("uxai-vendeur", "Vendeur Banc UX", new[] { Permissions.CatalogRead, Permissions.StockRead }),
    })
    {
        var existing = context.Users.FirstOrDefault(u => u.UserName == name);
        if (existing is null)
        {
            var user = new UserAccount(Guid.CreateVersion7(), name, display, now);
            user.SetPasswordHash(hashing.Hash(user, Password));
            foreach (var p in perms)
            {
                user.Grant(p);
            }

            context.Users.Add(user);
        }

        accounts.Add(new { key = name, userName = name, password = Password, displayName = display });
    }

    await context.SaveChangesAsync();
}

await File.WriteAllTextAsync(credsFile, JsonSerializer.Serialize(accounts));
using var http = new HttpClient { BaseAddress = api };
var login = await http.PostAsJsonAsync("/api/auth/login", new { userName = "uxai-admin", password = Password });
var token = JsonDocument.Parse(await login.Content.ReadAsStringAsync()).RootElement.GetProperty("accessToken").GetString();
http.DefaultRequestHeaders.Authorization = new("Bearer", token);

async Task<JsonElement> Send(HttpMethod method, string path, object body)
{
    using var request = new HttpRequestMessage(method, path) { Content = JsonContent.Create(body) };
    using var response = await http.SendAsync(request);
    var text = await response.Content.ReadAsStringAsync();
    if (!response.IsSuccessStatusCode)
    {
        Console.WriteLine($"{method} {path} → {(int)response.StatusCode} {text}");
        return default;
    }

    return text.Length == 0 ? default : JsonDocument.Parse(text).RootElement.Clone();
}
