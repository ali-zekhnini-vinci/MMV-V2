async Task<JsonElement> Send(HttpMethod method, string path, object body)
{
    using var request = new HttpRequestMessage(method, path) { Content = JsonContent.Create(body) };
    using var response = await http.SendAsync(request);
    var text = await response.Content.ReadAsStringAsync();
    if (!response.IsSuccessStatusCode)
    {
        Console.WriteLine($"{method} {path} → {(int)response.StatusCode} {text}");
        return default;
    }

    return text.Length == 0 ? default : JsonDocument.Parse(text).RootElement.Clone();
}

async Task Product(string family, string brand, string name, (string Label, string Color, string Ref, string Pa, string Coef, int Qty)[] variants, Guid supplier, string policy = "stocked")
{
    var body = new
    {
        requestId = Guid.NewGuid(), familyCode = family, brand, name,
        variants = variants.Select(v => new
        {
            variant = new { label = v.Label, stockPolicy = policy, attributes = new Dictionary<string, string> { ["color"] = v.Color } },
            suppliers = new[] { new { supplierId = supplier, supplierReference = v.Ref, isPrimary = true } },
            price = new { mode = "coefficient", purchasePriceHt = v.Pa, coefficient = v.Coef, expectedSalePriceHt = (decimal.Parse(v.Pa, System.Globalization.CultureInfo.InvariantCulture) * decimal.Parse(v.Coef, System.Globalization.CultureInfo.InvariantCulture)).ToString("F2", System.Globalization.CultureInfo.InvariantCulture) },
            stock = policy == "stocked" && v.Qty > 0 ? new { locationId = location, quantity = v.Qty } : null,
        }).ToArray(),
    };
    var created = await Send(HttpMethod.Post, "/api/catalog/products", body);
    Console.WriteLine(created.ValueKind == JsonValueKind.Undefined ? $"ÉCHEC {name}" : $"PRODUIT {name}");
}
