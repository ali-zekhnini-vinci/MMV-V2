async Task AddUser(string name, params string[] permissions)
{
    await using var scope = provider.CreateAsyncScope();
    var context = scope.ServiceProvider.GetRequiredService<MmvDbContext>();
    var user = new UserAccount(Guid.CreateVersion7(), $"{name}-{suffix}", $"Fictif {name}", DateTimeOffset.UtcNow);
    user.SetPasswordHash(scope.ServiceProvider.GetRequiredService<IPasswordHashing>().Hash(user, Password));
    foreach (var p in permissions) user.Grant(p);
    context.Users.Add(user);
    await context.SaveChangesAsync();
    users[name] = user.Id;
}

// ---------- HTTP ----------
var handler = new HttpClientHandler { ServerCertificateCustomValidationCallback = HttpClientHandler.DangerousAcceptAnyServerCertificateValidator };
async Task<HttpClient> Login(string name)
{
    var http = new HttpClient(handler, disposeHandler: false) { BaseAddress = api };
    var login = await http.PostAsJsonAsync("/api/auth/login", new { userName = $"{name}-{suffix}", password = Password });
    var token = (await login.Content.ReadFromJsonAsync<JsonElement>()).GetProperty("accessToken").GetString();
    http.DefaultRequestHeaders.Authorization = new AuthenticationHeaderValue("Bearer", token);
    return http;
}

static async Task<(HttpStatusCode Status, JsonElement Body, HttpResponseMessage Response)> Send(HttpClient http, HttpMethod method, string path, object? body = null)
{
    var request = new HttpRequestMessage(method, path) { Content = body is null ? null : JsonContent.Create(body) };
    var response = await http.SendAsync(request);
    var text = await response.Content.ReadAsStringAsync();
    return (response.StatusCode, text.Length == 0 ? default : JsonDocument.Parse(text).RootElement.Clone(), response);
}
