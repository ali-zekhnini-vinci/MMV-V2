async Task CleanupAsync(string suffix)
{
    await using var connection = new NpgsqlConnection(builder.ConnectionString);
    await connection.OpenAsync();
    await using var transaction = await connection.BeginTransactionAsync();
    async Task<int> Exec(string sql)
    {
        await using var command = new NpgsqlCommand(sql, connection, transaction);
        command.Parameters.AddWithValue("u", $"c04-%-{suffix}");
        command.Parameters.AddWithValue("l", $"Fictif-C04-{suffix}%");
        return await command.ExecuteNonQueryAsync();
    }

    await using (var identity = new NpgsqlCommand(IdentitySql, connection, transaction))
    {
        var value = (string)(await identity.ExecuteScalarAsync())!;
        if (!value.StartsWith("mmv_v2_test|mmv_v2_test|5433|", StringComparison.Ordinal))
        {
            throw new InvalidOperationException($"Cible de nettoyage refusée : {value}");
        }
    }

    const string users = "SELECT id FROM auth.users WHERE user_name LIKE @u";
    const string clients = "SELECT id FROM clients.clients WHERE last_name LIKE @l";
    string[] tables = ["prescription_revisions", "prescriptions", "measurement_revisions", "measurements"];
    foreach (var t in tables) await Exec($"ALTER TABLE optical.{t} DISABLE TRIGGER USER");
    var o = 0;
    foreach (var (records, revisions, column) in new[] { ("prescriptions", "prescription_revisions", "prescription_id"), ("measurements", "measurement_revisions", "measurement_id") })
    {
        o += await Exec($"DELETE FROM optical.{revisions} WHERE {column} IN (SELECT id FROM optical.{records} WHERE client_id IN ({clients}) OR created_by_user_id IN ({users}))");
        o += await Exec($"DELETE FROM optical.{records} WHERE client_id IN ({clients}) OR created_by_user_id IN ({users})");
    }

    foreach (var t in tables) await Exec($"ALTER TABLE optical.{t} ENABLE TRIGGER USER");
    await Exec("ALTER TABLE audit.events DISABLE TRIGGER trg_events_append_only");
    var e = await Exec($"DELETE FROM audit.events WHERE actor_user_id IN ({users}) OR client_id IN ({clients})");
    await Exec("ALTER TABLE audit.events ENABLE TRIGGER trg_events_append_only");
    var a = await Exec($"DELETE FROM appointments.appointments WHERE client_id IN ({clients}) OR created_by_user_id IN ({users})");
    var s = await Exec($"DELETE FROM auth.sessions WHERE user_id IN ({users})");
    var c = await Exec("DELETE FROM clients.clients WHERE last_name LIKE @l");
    var p = await Exec($"DELETE FROM auth.user_permissions WHERE user_id IN ({users})");
    var u = await Exec("DELETE FROM auth.users WHERE user_name LIKE @u");
    await using (var state = new NpgsqlCommand(
        "SELECT count(*) FROM pg_trigger t JOIN pg_class k ON k.oid = t.tgrelid JOIN pg_namespace n ON n.oid = k.relnamespace WHERE n.nspname IN ('optical','audit') AND NOT t.tgisinternal AND t.tgenabled <> 'O'",
        connection, transaction))
    {
        var disabled = (long)(await state.ExecuteScalarAsync())!;
        if (disabled != 0)
        {
            throw new InvalidOperationException($"{disabled} déclencheur(s) non réactivé(s) : transaction annulée.");
        }
    }

    await transaction.CommitAsync();
    Console.WriteLine($"NETTOYAGE {suffix} : {o} lignes optiques, {e} événements d'audit, {a} rendez-vous, {s} sessions, {c} clients, {p} droits, {u} comptes ; déclencheurs optical/audit tous actifs (O) ; RÉSIDUS mmv_v2_test = {await ResidueAsync()}");
}

async Task<Guid> AddUserAsync(string userName, string display, string password, params string[] permissions)
{
    await using var scope = services.CreateAsyncScope();
    var context = scope.ServiceProvider.GetRequiredService<MmvDbContext>();
    var user = new UserAccount(Guid.CreateVersion7(), userName, display, DateTimeOffset.UtcNow);
    user.SetPasswordHash(scope.ServiceProvider.GetRequiredService<IPasswordHashing>().Hash(user, password));
    foreach (var p in permissions) user.Grant(p);
    context.Users.Add(user);
    await context.SaveChangesAsync();
    return user.Id;
}

async Task<Guid> AddClientAsync(string first, string last, Guid author)
{
    await using var scope = services.CreateAsyncScope();
    var context = scope.ServiceProvider.GetRequiredService<MmvDbContext>();
    var client = Client.Create(Guid.CreateVersion7(), new ClientDetails(first, last, null, null, null, null, "Namur", null, null), author, DateTimeOffset.UtcNow);
    context.Clients.Add(client);
    await context.SaveChangesAsync();
    return client.Id;
}

    foreach (var p in paths)
    {
        var r = await plain.GetAsync(p);
        var t = await r.Content.ReadAsStringAsync();

            var ok = a.Status == HttpStatusCode.OK && (w.Status is HttpStatusCode.Created or HttpStatusCode.OK || (w.Status == HttpStatusCode.Conflict && Code(w.Body) == "client_archived"));
            if (!ok) bad++;
            outcomes.Add($"{(existing is null ? "création" : "rectif")}:{(int)w.Status}{(Code(w.Body) is { } code ? "/" + code : "")}+archive:{(int)a.Status}");
        }

        Check("9.1 écriture optique ‖ archivage (10 paires) : jamais de 500 ; écriture 201/200 OU 409 client_archived ; archivage 200", bad == 0, string.Join(" ", outcomes));
        var lateWrites = await ScalarAsync<long>("SELECT count(*) FROM optical.prescription_revisions r JOIN optical.prescriptions p ON p.id = r.prescription_id JOIN clients.clients c ON c.id = p.client_id WHERE c.id = ANY(@c) AND r.confirmed_at_utc > c.archived_at_utc", ("c", archivedClients.ToArray()));
        Check("9.2 constat : révisions confirmées horodatées APRÈS l'archivage (écriture commencée avant, sérialisable écriture→archivage ; réserve BL)", true, $"{lateWrites}");
        var after = await Send(full, HttpMethod.Post, "/api/optical/prescriptions", new { requestId = Guid.CreateVersion7(), clientId = archivedClients[0], confirm = true, content = Rx("-1,00") });
        Check("9.3 après archivage : nouvelle écriture -> 409 client_archived", after.Status == HttpStatusCode.Conflict && Code(after.Body) == "client_archived", $"{(int)after.Status} {Code(after.Body)}");
        var readArchived = await Send(reader, HttpMethod.Get, $"/api/optical/clients/{archivedClients[1]}/records");
        Check("9.4 client archivé : consultation 200", readArchived.Status == HttpStatusCode.OK, $"{(int)readArchived.Status}");
    });

    // ---------- 10. Désactivation d'un collaborateur simultanée à une confirmation de mesure ----------
    await Section("10", async () =>
    {
        var measurer = ids["mesureur"];
        var outcomes = new List<string>();
        var bad = 0;
        // Mesure confirmée AVANT la désactivation (déterministe) : sert à 10.4 / 10.5.
        var before = await Send(full, HttpMethod.Post, "/api/optical/measurements", new { requestId = Guid.CreateVersion7(), clientId, confirm = true, content = Mes(measurer, "+0,05") });
        string? keptId = before.Status == HttpStatusCode.Created ? before.Body.GetProperty("id").GetString() : null;
        var keptVersion = keptId is null ? 0 : V(before.Body);
        // Course : la désactivation est décalée de 0 à 45 ms pour couvrir les deux ordres (avant / après le contrôle d'éligibilité).
        for (var i = 0; i < 10; i++)
        {
            await SqlAsync("UPDATE auth.users SET is_active = true WHERE id = @u", ("u", measurer));
            var confirm = Send(full, HttpMethod.Post, "/api/optical/measurements", new { requestId = Guid.CreateVersion7(), clientId, confirm = true, content = Mes(measurer, $"+0,{i}5") });
            var delay = i * 5;
            var deactivate = Task.Run(async () => { await Task.Delay(delay); await SqlAsync("UPDATE auth.users SET is_active = false WHERE id = @u", ("u", measurer)); });
            await Task.WhenAll(confirm, deactivate);
            var r = confirm.Result;
            var ok = r.Status == HttpStatusCode.Created || (r.Status == HttpStatusCode.UnprocessableEntity && r.Text.Contains("not_eligible", StringComparison.Ordinal));
            if (!ok) bad++;
            outcomes.Add($"+{delay}ms:" + (r.Status == HttpStatusCode.Created ? "201" : $"{(int)r.Status}{(r.Text.Contains("not_eligible", StringComparison.Ordinal) ? "/not_eligible" : "")}"));
        }

        Check("10.1 confirmation de mesure ‖ désactivation du mesureur (10 paires décalées) : 201 OU 422 not_eligible, jamais de 500", bad == 0, string.Join(" ", outcomes));
        var after = await Send(full, HttpMethod.Post, "/api/optical/measurements", new { requestId = Guid.CreateVersion7(), clientId, confirm = true, content = Mes(measurer) });
        Check("10.2 mesureur désactivé : nouvelle confirmation -> 422 not_eligible, rien écrit", after.Status == HttpStatusCode.UnprocessableEntity && after.Text.Contains("not_eligible", StringComparison.Ordinal), $"{(int)after.Status}");
        var list = await Send(reader, HttpMethod.Get, "/api/optical/measurers");
        Check("10.3 mesureur désactivé absent de la liste des collaborateurs", !list.Text.Contains(measurer.ToString(), StringComparison.Ordinal));
        if (keptId is not null)
        {
            var read = await Send(reader, HttpMethod.Get, $"/api/optical/measurements/{keptId}");
            var cited = read.Body.GetProperty("citedUsers").EnumerateArray().Any(a => a.GetProperty("id").GetString() == measurer.ToString() && a.GetProperty("displayName").GetString() == "C04 Mesureur");
            Check("10.4 mesure confirmée avant désactivation : lisible, collaborateur désactivé nommé (citedUsers, O-04-48)", read.Status == HttpStatusCode.OK && cited, $"{(int)read.Status}");
            var rect = await Send(full, HttpMethod.Post, $"/api/optical/measurements/{keptId}/rectifications", new { requestId = Guid.CreateVersion7(), version = keptVersion, expectedRevisionNo = 1, reasonCode = "transcription_error" });
            var conf = await Send(full, HttpMethod.Post, $"/api/optical/measurements/{keptId}/draft/confirm", new { requestId = Guid.CreateVersion7(), version = V(rect.Body), content = Mes(measurer, "+0,99") });
            Check("10.5 rectification gardant le même collaborateur (désignation passée) -> 200", conf.Status == HttpStatusCode.OK, $"{(int)rect.Status}/{(int)conf.Status}");
        }
        else
        {
            Check("10.4/10.5 aucune mesure confirmée pendant la course : rectification avec collaborateur désactivé", false, "NOT_TESTED (aucune 201 obtenue)");
        }
    });

    // ---------- 11. Journal de l'API : aucune valeur, prescripteur, note ni nom ----------
    await Section("11", async () =>
    {
        await Task.Delay(1000);
        string log;
        await using (var stream = new FileStream(apiLog, FileMode.Open, FileAccess.Read, FileShare.ReadWrite))
        using (var readerLog = new StreamReader(stream)) log = await readerLog.ReadToEndAsync();
        var hits = new[] { secretPrescriber, secretNotes, SecretSphere, "Fictif-C04", "Réfraction magasin" }.Where(s => log.Contains(s, StringComparison.Ordinal)).ToList();
        // Blocs « fail: » : 23505 sur ux_*_creation_request_id = course de double création réémise, TRAITÉE par le service
        // (200, enregistrement existant) mais journalisée par EF Core (O-04-49) ; tout autre bloc est une erreur réelle.
        var blocks = log.Replace("\r\n", "\n").Split("\nfail: ").Skip(1).Select(b => b.Split(["\ninfo: ", "\nwarn: ", "\nfail: "], StringSplitOptions.None)[0]).ToList();
        var unique = blocks.Count(b => b.StartsWith("Microsoft.EntityFrameworkCore.Update[10000]", StringComparison.Ordinal) && b.Contains("SqlState: 23505", StringComparison.Ordinal) && b.Contains("ConstraintName: ux_", StringComparison.Ordinal) && b.Contains("_creation_request_id", StringComparison.Ordinal));
        var commands = blocks.Count(b => b.StartsWith("Microsoft.EntityFrameworkCore.Database.Command[20102]", StringComparison.Ordinal));
        var handled = commands == unique ? unique + commands : -1;
        Check("11.1 journal API : aucune valeur optique, prescripteur, note, catégorie ni nom de client", hits.Count == 0, hits.Count == 0 ? "" : string.Join(",", hits));
        Check("11.2 journal API : aucune erreur non traitée (hors 23505 creation_request_id traité, O-04-49)", blocks.Count == handled, $"fail={blocks.Count} dont 23505 traités={handled}");
    });

    var pass = results.Count(r => r.Status == "PASS");
    Console.WriteLine($"===== [{label}] TOTAL {pass}/{results.Count} PASS ; suffixe {suffix} ; client {clientId}");
    await CleanupAsync(suffix);
    return pass == results.Count;
}
